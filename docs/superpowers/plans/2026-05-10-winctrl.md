# WinCtrl 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现一个 Rust Python 模块，提供 Windows 显示器分辨率控制和音频设备管理功能。

**Architecture:** 基于 windows-rs crate 手动实现所有 Win32 API 调用，通过 PyO3 提供子模块化 Python API（display/audio），使用 Python 异常体系处理错误。

**Tech Stack:** Rust (edition 2021) + PyO3 0.22 + windows 0.58 + maturin 构建

**Spec:** `docs/superpowers/specs/2026-05-10-winctrl-design.md`

```rust
use pyo3::prelude::*;
use windows::Win32::Devices::Display::{
    EnumDisplayDevicesW, DISPLAY_DEVICEW, DISPLAY_DEVICE_ACTIVE,
};
use windows::Win32::Foundation::BOOL;

use super::{MonitorInfo, DisplayError};

/// 枚举所有显示器
pub fn list_monitors_impl() -> Result<Vec<MonitorInfo>, DisplayError> {
    let mut monitors = Vec::new();
    let mut i: u32 = 0;

    loop {
        let mut display_device = DISPLAY_DEVICEW {
            cb: std::mem::size_of::<DISPLAY_DEVICEW>() as u32,
            ..Default::default()
        };

        let result = EnumDisplayDevicesW(
            None,  // 查询当前显示适配器
            i,
            &mut display_device,
            DISPLAY_DEVICE_ACTIVE,
        );

        if !result.as_bool() {
            break;
        }

        let device_name = display_device.DeviceName.to_string();
        let friendly_name = display_device.DeviceString.to_string();
        let is_primary = (display_device.StateFlags & 1) != 0; // DISPLAY_DEVICE_PRIMARY_DEVICE

        monitors.push(MonitorInfo::new(
            device_name,
            friendly_name,
            is_primary,
        ));

        i += 1;
    }

    Ok(monitors)
}

/// Python 函数包装
#[pyfunction]
pub fn list_monitors(py: Python<'_>) -> PyResult<Vec<MonitorInfo>> {
    list_monitors_impl().map_err(|e| {
        DisplayError::new_err(e.to_string())
    })
}
```

```rust
use windows::Win32::Devices::Display::{
    EnumDisplaySettingsExW, ENUM_DISPLAY_SETTINGS_MODE,
};
use windows::Win32::UI::HiDpi::DEVMODEW;

pub fn get_current_resolution_impl(monitor_id: Option<String>) -> Result<Resolution, DisplayError> {
    let device_name = monitor_id.map(|s| windows::core::HSTRING::from(s));

    let mut dev_mode = DEVMODEW {
        dmSize: std::mem::size_of::<DEVMODEW>() as u16,
        ..Default::default()
    };

    let result = EnumDisplaySettingsExW(
        device_name.as_ref().map(|s| s.as_ref()).unwrap_or(None),
        ENUM_CURRENT_SETTINGS,
        &mut dev_mode,
        ENUM_DISPLAY_SETTINGS_MODE(0),
    );

    if !result.as_bool() {
        return Err(DisplayError::new_err("Failed to get current resolution"));
    }

    Ok(Resolution::new(
        dev_mode.dmPelsWidth,
        dev_mode.dmPelsHeight,
        dev_mode.dmDisplayFrequency,
    ))
}

#[pyfunction]
#[pyo3(signature = (monitor_id=None))]
pub fn get_current_resolution(py: Python<'_>, monitor_id: Option<String>) -> PyResult<Resolution> {
    get_current_resolution_impl(monitor_id).map_err(|e| {
        DisplayError::new_err(e.to_string())
    })
}
```

## Task 8: Display 模块 - set_resolution 实现

**Files:**
- Modify: `src/display/monitor.rs`

- [ ] **Step 1: 在 monitor.rs 中添加 set_resolution**

```rust
use windows::Win32::Devices::Display::{
    ChangeDisplaySettingsExW, CDS_UPDATEREGISTRY, CDS_TEST,
    DISP_CHANGE_SUCCESSFUL, DISP_CHANGE_BADMODE,
};
use std::cell::RefCell;

// 存储原始分辨率
thread_local! {
    static ORIGINAL_RESOLUTION: RefCell<Option<(String, Resolution)>> = RefCell::new(None);
}

pub fn set_resolution_impl(width: u32, height: u32, monitor_id: Option<String>) -> Result<(), DisplayError> {
    let device_name = monitor_id.map(|s| windows::core::HSTRING::from(s));

    // 先保存当前分辨率
    let current = get_current_resolution_impl(monitor_id.clone())?;
    let monitor_name = monitor_id.clone().unwrap_or_else(|| {
        list_monitors_impl()
            .ok()
            .and_then(|m| m.into_iter().find(|m| m.is_primary).map(|m| m.id))
            .unwrap_or_default()
    });
    ORIGINAL_RESOLUTION.with(|r| {
        *r.borrow_mut() = Some((monitor_name, current));
    });

    let mut dev_mode = DEVMODEW {
        dmSize: std::mem::size_of::<DEVMODEW>() as u16,
        dmPelsWidth: width,
        dmPelsHeight: height,
        dmFields: DM_PELSWIDTH | DM_PELSHEIGHT,
        ..Default::default()
    };

    // 先测试是否支持
    let test_result = ChangeDisplaySettingsExW(
        device_name.as_ref().map(|s| s.as_ref()).unwrap_or(None),
        Some(&mut dev_mode),
        None,
        CDS_TEST,
        None,
    );

    if test_result != DISP_CHANGE_SUCCESSFUL {
        return Err(DisplayError::new_err("Resolution not supported"));
    }

    // 应用设置
    let result = ChangeDisplaySettingsExW(
        device_name.as_ref().map(|s| s.as_ref()).unwrap_or(None),
        Some(&mut dev_mode),
        None,
        CDS_UPDATEREGISTRY,
        None,
    );

    if result != DISP_CHANGE_SUCCESSFUL {
        return Err(DisplayError::new_err("Failed to set resolution"));
    }

    Ok(())
}

const DM_PELSWIDTH: u32 = 0x00080000;
const DM_PELSHEIGHT: u32 = 0x00010000;

#[pyfunction]
#[pyo3(signature = (width, height, monitor_id=None))]
pub fn set_resolution(py: Python<'_>, width: u32, height: u32, monitor_id: Option<String>) -> PyResult<()> {
    set_resolution_impl(width, height, monitor_id).map_err(|e| {
        DisplayError::new_err(e.to_string())
    })
}
```

```rust
#[pyfunction]
#[pyo3(signature = (monitor_id=None))]
pub fn restore_resolution(py: Python<'_>, monitor_id: Option<String>) -> PyResult<()> {
    ORIGINAL_RESOLUTION.with(|r| {
        let stored = r.borrow().clone();
        if let Some((monitor_name, resolution)) = stored {
            // 恢复到保存的分辨率
            set_resolution_impl(resolution.width, resolution.height, Some(monitor_name))
                .map_err(|e| DisplayError::new_err(e.to_string()))?;
        } else {
            return Err(DisplayError::new_err("No original resolution saved"));
        }
        Ok(())
    })
}
```

```rust
pub fn list_devices_impl(device_type: String) -> Result<Vec<DeviceInfo>, AudioError> {
    let data_flow = match device_type.as_str() {
        "speaker" => eRender,
        "microphone" => eCapture,
        "all" => return list_all_devices_impl(),
        _ => return Err(AudioError::new_err("Invalid device_type")),
    };

    let enumerator: IMMDeviceEnumerator = CoCreateInstance(&MMDeviceEnumerator, None, CLSCTX_ALL)
        .map_err(|e| AudioError::new_err(format!("Failed to create enumerator: {}", e)))?;

    let collection: IMMDeviceCollection = enumerator.EnumAudioEndpoints(
        data_flow,
        DEVICE_STATEMASK_ALL,
    ).map_err(|e| AudioError::new_err(format!("Failed to enumerate: {}", e)))?;

    let count = collection.GetCount()
        .map_err(|e| AudioError::new_err(format!("Failed to get count: {}", e)))?;

    let mut devices = Vec::new();
    for i in 0..count {
        let device = collection.Item(i)
            .map_err(|e| AudioError::new_err(format!("Failed to get device: {}", e)))?;

        let id = device.GetId()
            .map_err(|e| AudioError::new_err(format!("Failed to get device ID: {}", e)))?;

        let state = device.GetState()
            .map_err(|e| AudioError::new_err(format!("Failed to get state: {}", e)))?;

        let state_str = match state {
            DEVICE_STATE_ACTIVE => "active",
            DEVICE_STATE_DISABLED => "disabled",
            DEVICE_STATE_UNPLUGGED => "unplugged",
            _ => "unknown",
        };

        let name = get_device_name(&device)?;

        let type_str = if data_flow == eRender { "speaker" } else { "microphone" };

        devices.push(DeviceInfo::new(
            id.to_string_lossy().to_string(),
            name,
            type_str.to_string(),
            state_str.to_string(),
        ));
    }

    Ok(devices)
}

    let speakers = list_devices_impl("speaker".to_string())?;

    Ok(speakers.into_iter().chain(mics.into_iter()).collect())
}

fn get_device_name(device: &IMMDevice) -> Result<String, AudioError> {

        .map_err(|e| AudioError::new_err(format!("Failed to get friendly name: {}", e)))?;

    Ok(value.to_string_lossy().to_string())
}
```

```rust
        .map_err(|e| AudioError::new_err(format!("Failed to create enumerator: {}", e)))?;

        .map_err(|e| AudioError::new_err(format!("Failed to get mute: {}", e)))?;
```

```rust
        state_str.to_string(),

        volume,
        muted.as_bool(),
```
